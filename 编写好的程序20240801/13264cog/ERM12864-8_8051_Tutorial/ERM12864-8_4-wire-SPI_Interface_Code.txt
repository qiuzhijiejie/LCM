void Write_Data(unsigned char dat)
{
    unsigned char idata i;
    RS=1;
	SCK=0;
	CS1=0;
	for(i=0;i<8;i++)
	{
	    SCK=0;
		_nop_();
		_nop_();
		SDA=dat&0x80;
		dat=dat<<1;
		_nop_();
		_nop_();
		SCK=1;
		_nop_();
		_nop_();
	}
	CS1=1;

	return;
}

{
    unsigned char idata i;
    RS=0;
	SCK=0;
	CS1=0;
	for(i=0;i<8;i++)
	{

		_nop_();
		_nop_();
		SDA=cmd&0x80;
		cmd=cmd<<1;
		_nop_();
		_nop_();
		SCK=1;
		_nop_();
		_nop_();
	}
	CS1=1;
	return;
}

void Display_Chess(unsigned char value)
{
    unsigned char i,j,k;
    for(i=0;i<0x08;i++)
	{
		Set_Page_Address(i);

        Set_Column_Address(0x00);

		for(j=0;j<0x10;j++)
		{
		    for(k=0;k<0x04;k++)
		        Write_Data(value);
		    for(k=0;k<0x04;k++)
		        Write_Data(~value);
		}
	}
    return;
}
